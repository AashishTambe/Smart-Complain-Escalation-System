┌─────────────────────────────────────────────────────────────────────────────────────────────────┐
│                           ESCALATION DECISION LOGIC FLOW                                         │
├─────────────────────────────────────────────────────────────────────────────────────────────────┤
│                                                                                                   │
│  EscalationJob runs [Scheduled]                                                                  │
│        │                                                                                         │
│        ▼                                                                                         │
│  Get active settings from DB (id=1)                                                             │
│        │                                                                                         │
│        ▼                                                                                         │
│  Is auto_escalation_enabled = 1?  ──[NO]──► Skip this complaint                                │
│        │                                                                                         │
│       [YES]                                                                                      │
│        │                                                                                         │
│        ▼                                                                                         │
│  Get complaint age (hours since last_status_change)                                             │
│        │                                                                                         │
│        ▼                                                                                         │
│  Get SLA for complaint priority (LOW/MEDIUM/HIGH)                                               │
│        │                                                                                         │
│        ▼                                                                                         │
│  Is age > SLA?  ──[NO]──► Not yet, wait for next run                                           │
│        │                                                                                         │
│       [YES]                                                                                      │
│        │                                                                                         │
│        ▼                                                                                         │
│  Is current_level < max_escalation_level?  ──[NO]──► Already at max, stop                      │
│        │                                                                                         │
│       [YES]                                                                                      │
│        │                                                                                         │
│        ▼                                                                                         │
│  ✓ ESCALATE COMPLAINT                                                                           │
│        │                                                                                         │
│        ├─► Increment current_level by 1                                                         │
│        ├─► Set status = 'ESCALATED'                                                             │
│        ├─► Find new assignee for new level                                                      │
│        ├─► Create notification (if enabled)                                                     │
│        ├─► Record history entry                                                                 │
│        └─► Update last_status_change to NOW()                                                   │
│                                                                                                   │

┌─────────────────────────────────────────────────────────────────────────────────────────────────┐
│                          COMMON SQL QUERIES FOR MAINTENANCE                                     │
├─────────────────────────────────────────────────────────────────────────────────────────────────┤
│                                                                                                   │
│  View all settings:                                                                             │
│  ► SELECT * FROM escalation_settings;                                                          │
│                                                                                                   │
│  View default settings:                                                                         │
│  ► SELECT * FROM escalation_settings WHERE id = 1;                                             │
│                                                                                                   │
│  Update SLA for HIGH priority:                                                                  │
│  ► UPDATE escalation_settings SET high_priority_sla_hours = 4 WHERE id = 1;                    │
│                                                                                                   │
│  Disable auto escalation (emergency):                                                           │
│  ► UPDATE escalation_settings SET auto_escalation_enabled = 0 WHERE id = 1;                    │
│                                                                                                   │
│  Enable auto escalation:                                                                        │
│  ► UPDATE escalation_settings SET auto_escalation_enabled = 1 WHERE id = 1;                    │
│                                                                                                   │
│  Check complaints due for escalation:                                                           │
│  ► SELECT c.id, c.priority, TIMESTAMPDIFF(HOUR, c.last_status_change, NOW()) as hours          │
│     FROM complaints c                                                                            │
│     WHERE c.status IN ('OPEN','IN_PROGRESS','ESCALATED')                                        │
│     AND c.current_level < 3 AND TIMESTAMPDIFF(HOUR, c.last_status_change, NOW()) > 6;          │
│                                                                                                   │
│  View escalation history:                                                                       │
│  ► SELECT * FROM complaint_history WHERE new_status = 'ESCALATED' ORDER BY changed_at DESC;   │
│                                                                                                   │
